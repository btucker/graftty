# Add non-git repos Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let Graftty add folders that are not git repositories — either by initializing a git repo in place or by registering the folder as a non-git project that renders one synthetic "main" worktree.

**Architecture:** Add `RepoEntry.isGitTracked` (decodes default `true`). A non-git repo is a normal `RepoEntry` with one normal `WorktreeEntry` whose `path == repo.path` and `branch == "main"`. A new `WorktreeDiscovery` facade dispatches: git-tracked → existing `GitWorktreeDiscovery.discover`, non-git → a synthetic single-entry list. Per-repo git polling fan-outs (`PRStatusStore`, `RemoteBranchStore`, `WorktreeStatsStore`) and git-only menu actions filter on the flag.

**Tech Stack:** Swift 5.x, Swift Testing (`@Test`, `@Suite`), SwiftUI/AppKit, the `git` CLI via `GitRunner`. Spec docs are regenerated by `scripts/generate-specs.py`.

**Design doc:** `docs/superpowers/specs/2026-05-11-add-non-git-repos-design.md`.

**Worktree:** This plan executes on the existing `add-non-git-repos` branch / worktree.

```swift
import Foundation

/// @spec PROJECT-1.0
/// Each repository entry shall record whether its on-disk path is
/// tracked by git. `false` means the folder has no `.git` and is
/// rendered through Graftty's UI as a single synthetic "main"
/// worktree at the repo path. See `WorktreeDiscovery` for how the
/// reconcile loop synthesizes the worktree list.
public struct RepoEntry: Codable, Sendable, Identifiable, Equatable {
    public let id: UUID
    public var path: String
    public var displayName: String
    public var isCollapsed: Bool
    public var worktrees: [WorktreeEntry]
    /// macOS URL bookmark for the repo folder, minted at add-time. Enables
    /// transparent recovery when the user renames or moves the folder in
    /// Finder (LAYOUT-4.5 .. LAYOUT-4.9). `nil` for entries decoded from a
    /// pre-LAYOUT-4.5 `state.json`; lazily minted on first successful path
    /// resolution after upgrade (LAYOUT-4.9).
    public var bookmark: Data?
    /// Whether the repo's on-disk path is tracked by git. Defaults to
    /// `true` for entries decoded from a pre-feature `state.json` so
    /// existing users upgrade in place (PROJECT-1.5). New entries added
    /// via "Add Without Git" persist as `false` and route through the
    /// `WorktreeDiscovery` synthetic path instead of `git worktree list`.
    public var isGitTracked: Bool

    public init(
        path: String,
        displayName: String,
        isCollapsed: Bool = false,
        worktrees: [WorktreeEntry] = [],
        bookmark: Data? = nil,
        isGitTracked: Bool = true
    ) {
        self.id = UUID()
        self.path = path
        self.displayName = displayName
        self.isCollapsed = isCollapsed
        self.worktrees = worktrees
        self.bookmark = bookmark
        self.isGitTracked = isGitTracked
    }

    private enum CodingKeys: String, CodingKey {
        case id, path, displayName, isCollapsed, worktrees, bookmark, isGitTracked
    }

    public init(from decoder: Decoder) throws {
        let container = try decoder.container(keyedBy: CodingKeys.self)
        self.id = try container.decode(UUID.self, forKey: .id)
        self.path = try container.decode(String.self, forKey: .path)
        self.displayName = try container.decode(String.self, forKey: .displayName)
        self.isCollapsed = try container.decodeIfPresent(Bool.self, forKey: .isCollapsed) ?? false
        self.worktrees = try container.decode([WorktreeEntry].self, forKey: .worktrees)
        self.bookmark = try container.decodeIfPresent(Data.self, forKey: .bookmark)
        self.isGitTracked = try container.decodeIfPresent(Bool.self, forKey: .isGitTracked) ?? true
    }
}
```

```swift
import Foundation

/// Dispatches worktree discovery for a `RepoEntry`. Git-tracked repos
/// shell out to `git worktree list --porcelain` (the existing
/// `GitWorktreeDiscovery.discover` path). Non-git repos return a
/// single synthetic worktree at the repo path with branch label
/// `"main"` (PROJECT-1.4), so the rest of the reconcile, sidebar,
/// and persistence code keeps working unchanged.
public enum WorktreeDiscovery {
    public static func discover(repo: RepoEntry) async throws -> [DiscoveredWorktree] {
        if repo.isGitTracked {
            return try await GitWorktreeDiscovery.discover(repoPath: repo.path)
        }
        return [DiscoveredWorktree(path: repo.path, branch: "main")]
    }
}
```

```swift
import Foundation

/// Initializes a git repository at `path` with an empty initial
/// commit. The empty commit gives the repo a real branch on
/// whatever `init.defaultBranch` is configured (`main` on modern
/// installs), so `git worktree list --porcelain` produces a named
/// branch when the Add Repository flow continues. Surfaces errors
/// as `GitInit.Error.gitFailed` so callers can present alerts in
/// the same style as `GitWorktreeRemove.Error`.
public enum GitInit {
    public enum Error: Swift.Error {
        case gitFailed(args: [String], stderr: String)
    }

    public static func run(at path: String) async throws {
        try await runStep(args: ["init"], at: path)
        try await runStep(
            args: ["commit", "--allow-empty", "-m", "Initial commit"],
            at: path
        )
    }

    private static func runStep(args: [String], at path: String) async throws {
        do {
            _ = try await GitRunner.run(args: args, at: path)
        } catch let err as CLIError {
            throw Error.gitFailed(args: args, stderr: err.stderr)
        }
    }
}
```

If `CLIError` does not expose `stderr` as a stored property in this codebase, replace the `catch` body with `throw Error.gitFailed(args: args, stderr: String(describing: err))`. Verify by reading `Sources/GrafttyKit/Util/CLIError.swift` (or the equivalent) before committing.

```swift
import AppKit
import GrafttyKit

/// UI-shaping helper for the three-button choice presented when the
/// user picks a folder that isn't a git repo (`GIT-1.5`). Keeps the
/// string table and the non-git-repo factory out of `MainWindow.swift`
/// so they can be unit-tested. The NSAlert itself is constructed by
/// the caller from these values.
enum AddRepositoryAlert {
    static let buttons = [
        "Initialize Git Repository",
        "Add Without Git",
        "Cancel",
    ]
    static let defaultButtonIndex = 0

    enum Choice {
        case initializeGit
        case addWithoutGit
        case cancel
    }

    /// Maps the modal-return of an NSAlert built from `buttons` to a
    /// `Choice`. `.alertFirstButtonReturn` corresponds to the default
    /// (`buttons[0]`), and so on. Any other return is treated as Cancel
    /// so the user's Esc / window-close gestures don't add anything.
    static func choice(for response: NSApplication.ModalResponse) -> Choice {
        switch response {
        case .alertFirstButtonReturn: return .initializeGit
        case .alertSecondButtonReturn: return .addWithoutGit
        default: return .cancel
        }
    }

    /// Constructs the non-git `RepoEntry` registered when the user
    /// chooses "Add Without Git" (`GIT-1.7`). The single synthetic
    /// worktree's path equals the repo path; its branch label is the
    /// literal string `"main"`.
    static func makeNonGitRepoEntry(
        atPath path: String,
        displayName: String,
        bookmark: Data?
    ) -> RepoEntry {
        RepoEntry(
            path: path,
            displayName: displayName,
            worktrees: [WorktreeEntry(path: path, branch: "main")],
            bookmark: bookmark,
            isGitTracked: false
        )
    }
}
```

```swift
import Foundation

/// Centralizes the "is this affordance shown?" predicates used by
/// the repo-row and worktree-row context menus and the Add Worktree
/// `+` button. Keeps the rules unit-testable and lets the views
/// stay focused on layout. PROJECT-1.1.
enum SidebarMenuVisibility {
    static func showsAddWorktree(repo: RepoEntry) -> Bool {
        repo.isGitTracked

    static func showsDeleteWorktree(worktree: WorktreeEntry, repo: RepoEntry) -> Bool {

        // there avoids a guaranteed error path. For non-git repos the
        // synthetic worktree's path always equals the repo path, so
        // this same predicate covers PROJECT-1.1 by construction.
        worktree.path != repo.path
    }
}
```

```swift
private func initializeGitRepositoryInPlace(_ repo: RepoEntry) {
    Task { @MainActor in
        do {
            try await GitInit.run(at: repo.path)

        appState.repos[idx].isGitTracked = true

        // Rediscover real worktrees and replace the synthetic entry.
        let updatedRepo = appState.repos[idx]
        let discovered: [DiscoveredWorktree]
        do {
            discovered = try await WorktreeDiscovery.discover(repo: updatedRepo)
        } catch {
            NSLog("[Graftty] initializeGitRepositoryInPlace: discover failed for %@: %@",
                  repo.path, String(describing: error))

        appState.repos[idx2].worktrees = discovered.map {
            WorktreeEntry(path: $0.path, branch: $0.branch)
        }

    }
}
```
